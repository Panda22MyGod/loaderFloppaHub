    end

RunService.RenderStepped:Connect(function()
    local t = tick()
    for _, data in ipairs(PulseData) do
        if data.label and data.label.Parent then
            local s = (math.sin(t * 2) + 1) / 2
            data.label.TextTransparency = data.base + s * 0.55
        end
    end
end)

local function bindClick(btn, callback)
    local lastFired = 0
    local function fire()
        local now = tick()
        if now - lastFired < 0.15 then return end
        lastFired = now
        if callback then callback() end
    end
    btn.MouseButton1Click:Connect(fire)
    btn.TouchTap:Connect(fire)
    btn.Activated:Connect(fire)
end

local Tabs = {}
local ActiveTab = nil

local function setTabActive(tab, active)
    TweenService:Create(tab.stroke, TweenInfo.new(0.2), {
        Color = active and BLUE_COLOR or GRAY_STROKE
    }):Play()
    TweenService:Create(tab.label, TweenInfo.new(0.2), {
        TextColor3 = active and BLUE_COLOR or INACTIVE_TXT
    }):Play()
end

local function registerTab(btn, label, stroke, onActivate)
    local tab = { btn = btn, label = label, stroke = stroke, onActivate = onActivate }
    table.insert(Tabs, tab)
    bindClick(btn, function()
        playClick()
        if ActiveTab then setTabActive(ActiveTab, false) end
        ActiveTab = tab
        setTabActive(tab, true)
        if onActivate then onActivate() end
    end)
    return tab
end

local function bindButton(btn, onClick)
    local originalColor = btn.BackgroundColor3

    btn.MouseButton1Down:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.08), {
            BackgroundColor3 = Color3.fromRGB(35, 35, 35)
        }):Play()
        playClick()
    end)
    btn.MouseButton1Up:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.08), {
            BackgroundColor3 = originalColor
        }):Play()
    end)
    bindClick(btn, onClick)
end

do
    local dragging = false
    local dragStart, startPos
    local targetPos  = G2L["2"].Position
    local currentPos = targetPos

    local function isInsideGui(input)
        local pos = input.Position
        local guiPos  = G2L["2"].AbsolutePosition
        local guiSize = G2L["2"].AbsoluteSize
        return pos.X >= guiPos.X and pos.X <= guiPos.X + guiSize.X
           and pos.Y >= guiPos.Y and pos.Y <= guiPos.Y + guiSize.Y
    end

    local function isOnResizeHandle(input)
        local handle = G2L["2"]:FindFirstChild("ResizeHandle")
        if not handle then return false end
        local pos = input.Position
        local hp  = handle.AbsolutePosition
        local hs  = handle.AbsoluteSize
        return pos.X >= hp.X and pos.X <= hp.X + hs.X
           and pos.Y >= hp.Y and pos.Y <= hp.Y + hs.Y
    end

    UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            if G2L["2"].Visible
            and isInsideGui(input)
            and not isOnResizeHandle(input)
            and not (_G.FloppaIsLocked and _G.FloppaIsLocked()) then
                dragging = true
                dragStart = input.Position
                startPos  = currentPos
            end
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            targetPos = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    RunService.RenderStepped:Connect(function(dt)
        if not G2L["2"] or not G2L["2"].Parent then return end
        local alpha = math.clamp(dt * 18, 0, 1)
        currentPos = UDim2.new(
            currentPos.X.Scale + (targetPos.X.Scale - currentPos.X.Scale) * alpha,
            currentPos.X.Offset + (targetPos.X.Offset - currentPos.X.Offset) * alpha,
            currentPos.Y.Scale + (targetPos.Y.Scale - currentPos.Y.Scale) * alpha,
            currentPos.Y.Offset + (targetPos.Y.Offset - currentPos.Y.Offset) * alpha
        )
        G2L["2"].Position = currentPos
    end)
end
